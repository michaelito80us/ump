### Technical Implementation

#### Plugin Discovery Script

```javascript
const fs = require('fs');
const path = require('path');

const categories = ['sports', 'phases', 'scheduling', 'seeding', 'tournaments'];
const plugins = [];

categories.forEach((category) => {
  const categoryPath = path.join('packages/plugins/src', category);
  if (fs.existsSync(categoryPath)) {
    const items = fs.readdirSync(categoryPath, { withFileTypes: true });
    items.forEach((item) => {
      if (item.isDirectory()) {
        plugins.push({
          id: item.name,
          category: category,
          path: path.join(categoryPath, item.name),
          generatedId: `${category}-${item.name}`,
        });
      }
    });
  }
});
```
