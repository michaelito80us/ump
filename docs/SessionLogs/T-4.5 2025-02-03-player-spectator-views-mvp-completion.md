### Subscription Architecture

```typescript
// useLeaderboardSubscription hook structure
const { data, loading, error, restart } = useSubscription(
  LEADERBOARD_UPDATED_SUBSCRIPTION,
  {
    variables: { tournamentId },
    skip: !enabled || !tournamentId,
    onError: (error) => console.error('Leaderboard subscription error:', error),
  }
);
```
